#!/bin/bash

END_HOUR=18
END_MIN=0

CUR_HOUR=$(date +%H)
CUR_MIN=$(date +%M)

CUR_TOTAL=$((CUR_HOUR*60 + CUR_MIN))
END_TOTAL=$((END_HOUR*60 + END_MIN))

REMAIN=$((END_TOTAL - CUR_TOTAL))

if [ $REMAIN -le 0 ]; then
  echo "Work day is already over"
else
  H=$((REMAIN / 60))
  M=$((REMAIN % 60))
  echo "Current time: $CUR_HOUR:$CUR_MIN"
  echo "Work day ends after $H hours and $M minutes"
fi

FILE=$1
WORD=$2

if [ -z "$FILE" ] || [ -z "$WORD" ]; then
  echo "Usage: count_word filename word"
  exit 1
fi

COUNT=$(grep -o -i "$WORD" "$FILE" | wc -l)
echo "The word '$WORD' appears $COUNT times in $FILE"

DIR=$1

if [ -z "$DIR" ]; then

  echo "Deleting $file"
  rm "$file"
done
